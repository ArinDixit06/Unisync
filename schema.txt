CREATE TABLE public.contacts (
  id uuid NOT NULL DEFAULT uuid_generate_v4(),
  user_id uuid NOT NULL,
  email_address text NOT NULL,
  display_name text,
  interaction_count integer DEFAULT 1,
  last_interaction_at timestamp with time zone DEFAULT now(),
  CONSTRAINT contacts_pkey PRIMARY KEY (id),
  CONSTRAINT contacts_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id)
);
CREATE TABLE public.drafts (

  user_id uuid NOT NULL,
  account_id uuid NOT NULL,
  to_list ARRAY NOT NULL,
  cc_list ARRAY,
  bcc_list ARRAY,
  subject text,
  body_html text,
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),
  to_list_enc text,
  cc_list_enc text,
  bcc_list_enc text,
  subject_enc text,
  body_html_enc text,
  CONSTRAINT drafts_pkey PRIMARY KEY (id),
  CONSTRAINT drafts_user_id_fkey FOREIGN KEY (user_id) REFERENCES public.users(id),
  CONSTRAINT drafts_account_id_fkey FOREIGN KEY (account_id) REFERENCES public.linked_accounts(id)
);
